Data class SecureDisasterMessage(
    val messageId: String,           // Unique UUID
    val timestamp: Long,             // Unix timestamp  
    val content: DisasterContent,    // Emergency data
    val senderDeviceId: String,      // Device identifier
    val signature: ByteArray,        // HMAC-SHA256 signature
    val hopCount: Int = 0,          // Track message propagation
    val messageHash: ByteArray       // SHA-256 of content



    
)
